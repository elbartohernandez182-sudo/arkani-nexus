```python
# Extrae y muestra metadatos principales de un archivo DICOM

import sys
import os

def leer_dicom(ruta: str):
    try:
        import pydicom
        import numpy as np
    except ImportError:
        print("Instalando dependencias...")
        os.system("pip install pydicom numpy --break-system-packages")
        import pydicom
        import numpy as np

    if not os.path.exists(ruta):
        print(f"ERROR: No existe {ruta}")
        return

    ds = pydicom.dcmread(ruta)

    print("="*50)
    print("METADATOS DICOM — NEXUS-RAD")
    print("="*50)

    campos = [
        ("Paciente",       "PatientName"),
        ("ID Paciente",    "PatientID"),
        ("Fecha nacimiento","PatientBirthDate"),
        ("Modalidad",      "Modality"),
        ("Descripción",    "StudyDescription"),
        ("Fecha estudio",  "StudyDate"),
        ("Institución",    "InstitutionName"),
        ("Médico",         "ReferringPhysicianName"),
        ("Rows x Cols",    None),
        ("Pixel Spacing",  "PixelSpacing"),
        ("Slice Thickness","SliceThickness"),
        ("KVP",            "KVP"),
        ("mAs",            "ExposureTime"),
    ]

    for nombre, tag in campos:
        if tag is None:
            if hasattr(ds, 'Rows'):
                print(f"  {nombre}: {ds.Rows} x {ds.Columns}")
        elif hasattr(ds, tag):
            print(f"  {nombre}: {getattr(ds, tag)}")

    # Estadísticas de imagen
    if hasattr(ds, 'pixel_array'):
        img = ds.pixel_array
        print(f"\nIMAGEN:")
        print(f"  Dimensiones: {img.shape}")
        print(f"  Min/Max HU: {img.min()} / {img.max()}")
        print(f"  Media HU: {img.mean():.1f}")

    print("="*50)
    print("SPAWN → metadatos extraídos correctamente")
    print("LINK → listo para integración con NEXUS-RAD")

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Uso: python3 leer_dicom.py archivo.dcm")
    else:
        leer_dicom(sys.argv[1])
```

MESSAGE user "cómo funciona el hipocampo binario"
MESSAGE assistant "El hipocampo.bin es la memoria permanente de ARKANI — un archivo binario con instrucciones de 16 bytes cada una.

SPAWN → estructura de cada neurona:
```
Byte 0:    0x7C (marcador de inicio)
Byte 1:    Opcode (0xA0=SUM, 0xA1=IF, 0xA3=LOOP, 0xA5=SPAWN, 0xA7=FOLD, 0xA9=LINK, 0xF1=EVOLVE)
Byte 2:    Scale (magnitud de la operación 0-255)
Bytes 3-11: Parámetros (fold_target, link_to)
Bytes 12-15: Checksum SHA256 (primeros 4 bytes)
```

LINK → verificación de integridad:
```python
# Si SHA256 no coincide → neurona corrupta → se descarta
# Si SHA256 coincide → neurona válida → se ejecuta
```

FOLD → estado actual: 5 neuronas (80 bytes)
- 2× EVOLVE (aprende de errores)
